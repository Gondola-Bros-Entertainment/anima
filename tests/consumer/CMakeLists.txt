cmake_minimum_required(VERSION 3.25)
project(AnimaConsumer LANGUAGES CXX)
include(CTest)
set(ANIMA_SOURCE_DIR "" CACHE PATH "Anima source checkout, pinned by the consuming project")
set(CONSUMER_MODE "assets" CACHE STRING "Independent public target configuration")
if(NOT EXISTS "${ANIMA_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Supply -DANIMA_SOURCE_DIR=/path/to/pinned/anima")
endif()
if(NOT CONSUMER_MODE MATCHES "^(core|audio_output|input_sdl|assets|desktop|ui|physics|physics_scene|ui_documents|physics2d|physics2d_scene|runtime)$")
    message(FATAL_ERROR "Unknown CONSUMER_MODE: ${CONSUMER_MODE}")
endif()

# Normal variables affect this dependency without changing the parent's test option.
set(ANIMA_BUILD_TOOLS OFF)
set(ANIMA_BUILD_TESTS OFF)
set(ANIMA_BUILD_DESKTOP OFF)
set(ANIMA_BUILD_AUDIO_OUTPUT OFF)
set(ANIMA_BUILD_ASSETS OFF)
set(ANIMA_BUILD_UI OFF)
set(ANIMA_BUILD_INPUT_SDL OFF)
if(CONSUMER_MODE STREQUAL "input_sdl")
    set(ANIMA_BUILD_INPUT_SDL ON)
endif()
if(CONSUMER_MODE STREQUAL "audio_output")
    set(ANIMA_BUILD_AUDIO_OUTPUT ON)
    # This consumer also builds against prepared SDL source on offline builders.
    # Its selected runtime needs no SDL graphics/camera subsystem.
    set(SDL_VIDEO OFF CACHE BOOL "" FORCE)
    set(SDL_RENDER OFF CACHE BOOL "" FORCE)
    set(SDL_GPU OFF CACHE BOOL "" FORCE)
    set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
    set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
endif()
set(ANIMA_BUILD_UI_DOCUMENTS OFF)
set(ANIMA_BUILD_PHYSICS OFF)
set(ANIMA_BUILD_PHYSICS2D OFF)
if(NOT CONSUMER_MODE MATCHES "^(core|audio_output|input_sdl|physics|physics2d|ui_documents)$")
    set(ANIMA_BUILD_ASSETS ON)
endif()
if(CONSUMER_MODE MATCHES "^(desktop|ui)$")
    set(ANIMA_BUILD_DESKTOP ON)
endif()
if(CONSUMER_MODE STREQUAL "ui")
    set(ANIMA_BUILD_UI ON)
endif()
if(CONSUMER_MODE MATCHES "^(physics|physics_scene|runtime)$")
    set(ANIMA_BUILD_PHYSICS ON)
endif()
if(CONSUMER_MODE STREQUAL "ui_documents")
    set(ANIMA_BUILD_UI_DOCUMENTS ON)
endif()
if(CONSUMER_MODE MATCHES "^(physics2d|runtime)")
    enable_language(C)
    set(ANIMA_BUILD_PHYSICS2D ON)
endif()
add_subdirectory("${ANIMA_SOURCE_DIR}" anima)
if(NOT ANIMA_BUILD_INPUT_SDL AND TARGET anima::input_sdl)
    message(FATAL_ERROR "Consumer imported optional SDL input dependency")
endif()
if(NOT ANIMA_BUILD_AUDIO_OUTPUT AND TARGET anima::audio_output)
    message(FATAL_ERROR "Consumer imported optional SDL audio output dependency")
endif()
if(NOT ANIMA_BUILD_DESKTOP AND TARGET anima::desktop)
    message(FATAL_ERROR "Consumer imported optional desktop dependency")
endif()
if(NOT ANIMA_BUILD_PHYSICS2D AND TARGET box2d)
    message(FATAL_ERROR "Consumer imported optional Box2D dependency")
endif()
if(NOT ANIMA_BUILD_PHYSICS AND TARGET Jolt)
    message(FATAL_ERROR "Consumer imported optional physics dependency")
endif()
foreach(forbidden anima anima_headless anima_inspect anima_core_tests anima_asset_tests anima_animation_tests anima_scene_tests anima_scene_validation_tests anima_docs Doxygen::doxygen)
    if(TARGET "${forbidden}")
        message(FATAL_ERROR "Library consumption created an unwanted tool/test: ${forbidden}")
    endif()
endforeach()
if(NOT CONSUMER_MODE MATCHES "^(audio_output|desktop|ui)$" AND TARGET SDL3::SDL3)
    message(FATAL_ERROR "Consumer imported optional SDL runtime dependency")
endif()
if(NOT CONSUMER_MODE MATCHES "^(desktop|ui)$" AND TARGET Vulkan::Vulkan)
    message(FATAL_ERROR "Headless consumer imported a graphics dependency")
endif()
if(CONSUMER_MODE MATCHES "^(core|audio_output|input_sdl|physics|physics2d|assets|physics_scene|physics2d_scene|runtime)$" AND (TARGET RmlUi::Core OR TARGET Freetype::Freetype))
    message(FATAL_ERROR "Consumer imported optional UI dependency")
endif()
if(CONSUMER_MODE MATCHES "^(core|audio_output|input_sdl|ui_documents|physics|physics2d)$" AND TARGET anima::assets)
    message(FATAL_ERROR "Core-only consumer imported asset dependencies")
endif()

add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE anima::core)
target_compile_features(consumer PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(consumer PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(NOT CONSUMER_MODE MATCHES "^(core|audio_output|input_sdl|physics|physics2d|ui_documents)$")
    if(CONSUMER_MODE STREQUAL "assets")
        target_link_libraries(consumer PRIVATE anima::assets)
    endif()
    target_compile_definitions(consumer PRIVATE CONSUMER_ASSETS=1)
endif()
if(CONSUMER_MODE STREQUAL "audio_output")
    target_link_libraries(consumer PRIVATE anima::audio_output)
    target_compile_definitions(consumer PRIVATE CONSUMER_AUDIO_OUTPUT=1)
    anima_copy_sdl_runtime(consumer)
endif()
if(CONSUMER_MODE MATCHES "^(desktop|ui)$")
    if(NOT TARGET SDL3::SDL3)
        message(FATAL_ERROR "Desktop dependency did not expose SDL to the consuming directory")
    endif()
    if(CONSUMER_MODE STREQUAL "desktop")
        target_link_libraries(consumer PRIVATE anima::desktop)
    endif()
    target_compile_definitions(consumer PRIVATE CONSUMER_DESKTOP=1)
    anima_copy_sdl_runtime(consumer)
    anima_copy_sdl_runtime(consumer) # Idempotent for application helpers wrapping it.
endif()
if(CONSUMER_MODE STREQUAL "ui")
    target_link_libraries(consumer PRIVATE anima::ui)
    target_compile_definitions(consumer PRIVATE CONSUMER_UI=1)
endif()
if(CONSUMER_MODE MATCHES "^(physics|physics_scene|runtime)$")
    target_link_libraries(consumer PRIVATE anima::physics)
    target_compile_definitions(consumer PRIVATE CONSUMER_PHYSICS=1)
endif()
if(CONSUMER_MODE STREQUAL "physics_scene")
    target_link_libraries(consumer PRIVATE anima::physics_scene)
    target_compile_definitions(consumer PRIVATE CONSUMER_PHYSICS_SCENE=1)
endif()
if(CONSUMER_MODE STREQUAL "ui_documents")
    target_link_libraries(consumer PRIVATE anima::ui_documents RmlUi::Core)
    target_compile_definitions(consumer PRIVATE CONSUMER_UI_DOCUMENTS=1)
endif()
if(CONSUMER_MODE MATCHES "^(physics2d|runtime)")
    target_link_libraries(consumer PRIVATE anima::physics2d)
    target_compile_definitions(consumer PRIVATE CONSUMER_PHYSICS2D=1)
endif()
if(CONSUMER_MODE STREQUAL "physics2d_scene")
    target_link_libraries(consumer PRIVATE anima::physics2d_scene)
    target_compile_definitions(consumer PRIVATE CONSUMER_PHYSICS2D_SCENE=1)
endif()
if(CONSUMER_MODE STREQUAL "runtime")
    target_link_libraries(consumer PRIVATE anima::physics_scene anima::physics2d_scene)
    target_compile_definitions(consumer PRIVATE CONSUMER_RUNTIME=1)
endif()
if(BUILD_TESTING)
    add_test(NAME consumer_public_api COMMAND consumer)
    if(CONSUMER_MODE STREQUAL "audio_output")
        set_tests_properties(consumer_public_api PROPERTIES ENVIRONMENT "SDL_AUDIO_DRIVER=dummy" TIMEOUT 20)
    endif()
else()
    message(FATAL_ERROR "Anima changed the consumer's BUILD_TESTING option")
endif()

if(CONSUMER_MODE STREQUAL "input_sdl")
    target_link_libraries(consumer PRIVATE anima::input_sdl)
    target_include_directories(consumer SYSTEM PRIVATE "${ANIMA_INPUT_SDL_INCLUDE_DIR}")
    target_compile_definitions(consumer PRIVATE CONSUMER_INPUT_SDL=1)
endif()
